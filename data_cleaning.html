Cleaning

# Convert climate YEAR and DOY into dates
climate_df["date"] = (
    pd.to_datetime(climate_df["YEAR"].astype(str) + "-01-01")
    + pd.to_timedelta(climate_df["DOY"] - 1, unit="D")
)

# Convert dengue dates
disease_df["weekdate"] = pd.to_datetime(disease_df["weekdate"])

# Sort both datasets
climate_df = climate_df.sort_values("date")
disease_df = disease_df.sort_values("weekdate")

# Replace NASA missing values
cols = ["T2M", "T2M_MAX", "T2M_MIN", "PRECTOTCORR", "RH2M"]

climate_df[cols] = climate_df[cols].replace(-999, np.nan)

climate_aligned = pd.merge_asof(
    climate_df,
    disease_df[["weekdate"]],
    left_on="date",
    right_on="weekdate",
    direction="backward"
)

climate_weekly = (
    climate_aligned
    .groupby("weekdate")
    .agg(
        temp_avg=("T2M", "mean"),
        temp_max=("T2M_MAX", "max"),
        temp_min=("T2M_MIN", "min"),
        precip_total=("PRECTOTCORR", lambda x: x.sum(min_count=1)),
        humidity_avg=("RH2M", "mean")
    )
    .reset_index()
)

full_df = disease_df.merge(
    climate_weekly,
    on="weekdate",
    how="left",
    validate="one_to_one"

full_df = full_df.sort_values("weekdate").reset_index(drop=True)
