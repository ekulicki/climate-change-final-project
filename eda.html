# Cases per year

full_df["weekdate"] = pd.to_datetime(full_df["weekdate"])

plt.figure(figsize=(14, 5))

sns.lineplot(
    data=full_df,
    x="weekdate",
    y="conf_cases"
)

plt.title("Weekly Confirmed Dengue Cases in Puerto Rico (1986–2024)")
plt.xlabel("Year")
plt.ylabel("Confirmed Dengue Cases")
plt.tight_layout()
plt.show()
annual_cases = (
    full_df.groupby("year")["conf_cases"]
    .sum()
    .reset_index()
)

plt.figure(figsize=(14, 5))

sns.barplot(
    data=annual_cases,
    x="year",
    y="conf_cases",
    color="steelblue"
)

plt.title("Annual Confirmed Dengue Cases in Puerto Rico")
plt.xlabel("Year")
plt.ylabel("Total Confirmed Cases")
plt.xticks(rotation=90)
plt.tight_layout()
plt.show()
# Seasonal trends

weekly_avg = (
    full_df.groupby("week")["conf_cases"]
    .mean()
    .reset_index()
)

plt.figure(figsize=(12, 5))

sns.lineplot(
    data=weekly_avg,
    x="week",
    y="conf_cases"
)

plt.title("Average Weekly Dengue Cases by Week of Year")
plt.xlabel("Week of Year")
plt.ylabel("Average Confirmed Cases")
plt.tight_layout()
plt.show()
fig, axes = plt.subplots(3, 1, figsize=(14, 12))

sns.lineplot(
    data=full_df,
    x="weekdate",
    y="temp_avg",
    ax=axes[0]
)
axes[0].set_title("Weekly Average Temperature")

sns.lineplot(

climate_vars = ["temp_avg", "precip_total", "humidity_avg"]

fig, axes = plt.subplots(1, 3, figsize=(16, 5))

for ax, var in zip(axes, climate_vars):
    sns.scatterplot(
        data=full_df,
        x=var,
        y="conf_cases",
        alpha=0.4,
        ax=ax
    )

    ax.set_title(f"Dengue Cases vs. {var}")
    ax.set_ylabel("Confirmed Dengue Cases")

plt.tight_layout()
plt.show()
# Heatmap
cols = [
    "conf_cases",
    "temp_avg",
    "temp_max",
    "temp_min",
    "precip_total",
    "humidity_avg"
]

corr_matrix = full_df[cols].corr()

plt.figure(figsize=(9, 7))

sns.heatmap(
    corr_matrix,
    annot=True,
    cmap="coolwarm",
    center=0,
    fmt=".2f"
)

plt.title("Correlation Between Climate Variables and Dengue Cases")
plt.tight_layout()
plt.show()
