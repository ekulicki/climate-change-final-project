Set Up Hyperparameters

from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from xgboost import XGBRegressor

features = [
    "temp_avg",
    "temp_max",
    "temp_min",
    "precip_total",
    "humidity_avg"
]

horizons = [1, 2, 4, 8]

models = {
    "Linear Regression": LinearRegression(),
    "Random Forest": RandomForestRegressor(
        n_estimators=200,
        random_state=42,
        n_jobs=-1
    ),
    "XGBoost": XGBRegressor(
        n_estimators=200,
        max_depth=3,
        learning_rate=0.05,
        objective="reg:squarederror",
        random_state=42,
        n_jobs=-1
    )
}

Train Models: Linear Regression, Random Forest, XGBoost

from sklearn.base import clone

results = []
trained_models = {}

for horizon in horizons:

    target = f"cases_{horizon}wk_later"

    # Define chronological splits
    # Exclude the final 'horizon' weeks of training
    # so training targets don't extend into validation
    train = full_df[
    (full_df["year"] <= 2015) &
    ~(
        (full_df["year"] == 2015) &
        (full_df["week"] > 52 - horizon)
    )
]   
    val = full_df[
        (full_df["year"] >= 2016) &
        (full_df["year"] <= 2019) &
        ~(
            (full_df["year"] == 2019) &
            (full_df["week"] > 52 - horizon)
        )
    ]

    X_train = train[features]
    y_train = train[target]

    X_val = val[features]
    y_val = val[target]

    for name, model in models.items():

        fitted_model = clone(model)
        fitted_model.fit(X_train, y_train)
        predictions = fitted_model.predict(X_val)

        results.append({
            "Model": name,
            "Horizon": horizon,
            "MAE": mean_absolute_error(y_val, predictions),
            "RMSE": np.sqrt(mean_squared_error(y_val, predictions)),
            "R2": r2_score(y_val, predictions)
        })

        trained_models[(name, horizon)] = fitted_model

results_df = pd.DataFrame(results)

Long Short-Term Memory (LSTM) Neural Network

from sklearn.preprocessing import StandardScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Input, LSTM, Dense, Dropout
from tensorflow.keras.callbacks import EarlyStopping

sequence_length = 8

scaler = StandardScaler()

train_mask = full_df["year"] <= 2015

scaler.fit(full_df.loc[train_mask, features])

X_scaled = scaler.transform(full_df[features])

def create_sequences(X, y, years, weeks, sequence_length):
    sequences = []
    targets = []
    input_years = []
    input_weeks = []

    for i in range(sequence_length - 1, len(X)):
        sequences.append(X[i-sequence_length+1:i+1])
        targets.append(y[i])
        input_years.append(years[i])
        input_weeks.append(weeks[i])

    return (
        np.array(sequences),
        np.array(targets),
        np.array(input_years),
        np.array(input_weeks)
    )

Train LSTM Model

lstm_results = []
lstm_models = {}

for horizon in horizons:

    target = f"cases_{horizon}wk_later"

    X_seq, y_seq, years, weeks = create_sequences(
        X_scaled,
        full_df[target].to_numpy(),
        full_df["year"].to_numpy(),
        full_df["week"].to_numpy(),
        sequence_length
    )
    train_mask = (
        (years <= 2015) &
        ~((years == 2015) & (weeks > 52 - horizon))
    )
    val_mask = (
        (years >= 2016) &
        (years <= 2019) &
        ~((years == 2019) & (weeks > 52 - horizon))
    )

    X_train, y_train = X_seq[train_mask], y_seq[train_mask]
    X_val, y_val = X_seq[val_mask], y_seq[val_mask]

    model = Sequential([
        Input(shape=(sequence_length, len(features))),
        LSTM(16),
        Dropout(0.2),
        Dense(1)
    ])

    model.compile(
        optimizer="adam",
        loss="mse"
    )

    early_stop = EarlyStopping(
        monitor="val_loss",
        patience=10,
        restore_best_weights=True
    )

    model.fit(
        X_train,
        y_train,
        validation_data=(X_val, y_val),
        epochs=100,
        batch_size=32,
        callbacks=[early_stop],
        verbose=0
    )

    predictions = model(X_val, training=False).numpy().flatten()

    lstm_results.append({
        "Model": "LSTM",
        "Horizon": horizon,
        "MAE": mean_absolute_error(y_val, predictions),
        "RMSE": np.sqrt(mean_squared_error(y_val, predictions)),
        "R2": r2_score(y_val, predictions)
    })

    lstm_models[horizon] = model

Comparing Against a Baseline

# Baseline model = predict the training-period mean

baseline_results = []

for horizon in horizons:

    target = f"cases_{horizon}wk_later"

    train = full_df[
        (full_df["year"] <= 2015) &
        ~(
            (full_df["year"] == 2015) &
            (full_df["week"] > 52 - horizon)
        )
    ]

    val = full_df[
        (full_df["year"].between(2016, 2019)) &
        ~(
            (full_df["year"] == 2019) &
            (full_df["week"] > 52 - horizon)
        )
    ]

    baseline_prediction = train[target].mean()

    predictions = np.full(len(val), baseline_prediction)

    baseline_results.append({
        "Model": "Baseline (Train Mean)",
        "Horizon": horizon,
        "MAE": mean_absolute_error(val[target], predictions),
        "RMSE": np.sqrt(mean_squared_error(val[target], predictions)),
        "R2": r2_score(val[target], predictions)
    })

baseline_results_df = pd.DataFrame(baseline_results)
